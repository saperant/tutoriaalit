% Yhteiset LaTeX-tyylit tutoriaalien PDF-versioille.

% Pitkät koodirivit katkaistaan eikä niitä päästetä marginaaliin.
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{
  breaklines,breakanywhere,commandchars=\\\{\}}

% Koodilohkoille vaalea tausta ja kehys.
\usepackage[most]{tcolorbox}
\definecolor{koodintausta}{HTML}{F7F7F5}
\definecolor{koodinreuna}{HTML}{D8D8D4}
\renewenvironment{Shaded}{%
  \begin{tcolorbox}[breakable, enhanced, colback=koodintausta,
    colframe=koodinreuna, boxrule=0.4pt, arc=1pt,
    left=4pt, right=4pt, top=2pt, bottom=2pt]}{\end{tcolorbox}}

% Lainauslohkoja käytetään huomautuksiin.
\usepackage{etoolbox}
\definecolor{huomreuna}{HTML}{4A7AB0}
\renewenvironment{quote}{%
  \begin{tcolorbox}[breakable, enhanced, colback=white,
    colframe=white, borderline west={2pt}{0pt}{huomreuna},
    left=8pt, right=0pt, top=1pt, bottom=1pt, sharp corners]}%
  {\end{tcolorbox}}

% Otsikot hieman hillitymmiksi.
\addtokomafont{disposition}{\rmfamily}
\addtokomafont{chapter}{\Huge}

% Taulukot eivät saa ylittää tekstialuetta.
\usepackage{booktabs}

% Ohjelmien tulosteet: valkoinen laatikko, harmaa reuna ja otsikko.
\definecolor{tulostereuna}{HTML}{9A9A94}
\newtcolorbox{tulostelaatikko}{breakable, enhanced jigsaw,
  colback=white, colframe=tulostereuna, boxrule=0.4pt, arc=1pt,
  left=4pt, right=4pt, top=2pt, bottom=2pt,
  overlay unbroken and first={\node[anchor=north east, inner sep=2pt,
    font=\scriptsize\sffamily\color{tulostereuna}]
    at (frame.north east) {tuloste};}}
\DefineVerbatimEnvironment{tulosteverbatim}{Verbatim}{breaklines,
  breakanywhere, fontsize=\small}
\newenvironment{tuloste}{\VerbatimEnvironment
  \begin{tulostelaatikko}\begin{tulosteverbatim}}%
  {\end{tulosteverbatim}\end{tulostelaatikko}}
