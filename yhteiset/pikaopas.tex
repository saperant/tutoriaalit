\RedeclareSectionCommand[beforeskip=4pt,afterskip=1pt,font=\normalsize\rmfamily\bfseries]{subsection}
\setlength{\columnseprule}{0.3pt}
\makeatletter
\renewcommand{\maketitle}{%
  {\noindent\Large\bfseries\@title\par}
  {\noindent\small\@subtitle\par}\vspace{4pt}}
\makeatother
\tcbset{before skip=2pt, after skip=3pt}

% Kaksipalstaisessa tilassa longtable ei ole sallittu. Pikaoppaan
% taulukot ovat lyhyitä, joten ne ladotaan tavallisina taulukoina.
% Pandoc kirjoittaa otsakkeen ja alatunnisteen (\bottomrule) ennen rivejä;
% alatunniste ja mahdollinen toistettu otsake ohitetaan, ja alaviiva
% piirretään taulukon loppuun.
\usepackage{longtable}
\renewenvironment{longtable}[1][]{\par\noindent\begin{tabular}}%
  {\bottomrule\end{tabular}\par}
\long\def\endfirsthead#1\endhead{\endhead}
\long\def\endhead#1\endlastfoot{}
